//Remove a time range from the middle of the video (re-encodes)
//CUT_START = 90   (seconds) - where removal begins
//CUT_END   = 120  (seconds) - where removal ends
//CUT_END must be greater than CUT_START and shorter than the video length

ffmpeg -y -i "{input}.*" -filter_complex "[0:v]trim=start=0:end=90,setpts=PTS-STARTPTS[v1];[0:a]atrim=start=0:end=90,asetpts=PTS-STARTPTS[a1];[0:v]trim=start=120,setpts=PTS-STARTPTS[v2];[0:a]atrim=start=120,asetpts=PTS-STARTPTS[a2];[v1][a1][v2][a2]concat=n=2:v=1:a=1[out]" -map "[out]" -c:v libx264 -preset veryfast -crf 23 -pix_fmt yuv420p -c:a aac -b:a 128k "{output}.mp4"
